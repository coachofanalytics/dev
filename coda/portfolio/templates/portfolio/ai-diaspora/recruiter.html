AI Cultural Matching API

@api_view(['POST'])
@permission_classes([IsAuthenticated])
def find_cultural_matches(request):
    """Find cultural matches using AI"""
    try:
        # Extract user cultural profile
        user_profile = extract_cultural_profile(request.user)
        
        # Get AI model prediction
        matches = ml_service.find_cultural_matches(
            user_profile=user_profile,
            limit=request.data.get('limit', 10)
        )
        
        # Store match results
        for match in matches:
            CulturalMatch.objects.create(
                profile1=request.user.profile,
                profile2_id=match['profile_id'],
                match_score=match['score'],
                compatibility_factors=match['factors']
            )
        
        return Response({
            'matches': matches,
            'total_found': len(matches),
            'model_version': ml_service.get_version()
        })
    except Exception as e:
        return Response({'error': str(e)}, status=500)
