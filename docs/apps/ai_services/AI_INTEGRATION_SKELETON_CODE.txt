"""
AI Insight Service

Centralized service for all AI-generated insights, explanations, and coaching text.

CRITICAL RULES:
- This service generates TEXT EXPLANATIONS ONLY.
- All numeric values (pay amounts, compliance percentages, career levels) must be passed in
  from deterministic services (PayCalculationService, ReleaseEngine, ComplianceCalculator, etc.).
- This service NEVER calculates financial amounts, compliance status, or career progression.
- This service is READ-ONLY over deterministic data and WRITE-ONLY for explanations/coaching text.
"""

import logging
from typing import Dict, Any, Optional, List
from django.contrib.auth import get_user_model

# Lazy imports to avoid circular dependencies
try:
    from ai_services.services.ai_service_facade import ai_service_facade
except ImportError:
    ai_service_facade = None

try:
    from ai_services.registry.prompt_template_registry import PromptTemplateRegistry
    from ai_services.registry.output_schema_registry import OutputSchemaRegistry
    from ai_services.services.rag_retrieval_service import RAGRetrievalService
except ImportError:
    PromptTemplateRegistry = None
    OutputSchemaRegistry = None
    RAGRetrievalService = None

logger = logging.getLogger(__name__)
User = get_user_model()


class AIInsightService:
    """
    Centralized service for all AI-generated insights, explanations, and coaching.
    
    This service uses PromptTemplateRegistry, OutputSchemaRegistry, and RAGRetrievalService
    to generate personalized, context-aware text for employees and managers.
    
    All numeric values must be provided by deterministic services - this service only generates text.
    """
    
    def __init__(
        self,
        prompt_registry: Optional[PromptTemplateRegistry] = None,
        output_registry: Optional[OutputSchemaRegistry] = None,
        rag_service: Optional[RAGRetrievalService] = None,
        ai_facade: Optional[Any] = None
    ):
        """
        Initialize AIInsightService.
        
        Args:
            prompt_registry: Optional PromptTemplateRegistry instance
            output_registry: Optional OutputSchemaRegistry instance
            rag_service: Optional RAGRetrievalService instance
            ai_facade: Optional AIServiceFacade instance (defaults to global ai_service_facade)
        """
        self.logger = logger
        self.prompt_registry = prompt_registry or (PromptTemplateRegistry() if PromptTemplateRegistry else None)
        self.output_registry = output_registry or (OutputSchemaRegistry() if OutputSchemaRegistry else None)
        self.rag_service = rag_service or (RAGRetrievalService() if RAGRetrievalService else None)
        self.ai_facade = ai_facade or ai_service_facade
        
        if not self.ai_facade:
            self.logger.warning("AIServiceFacade not available - AI insights will be disabled")
    
    def generate_pay_explanation(
        self,
        employee: User,
        pay_breakdown: Dict[str, Any],
        compliance_data: Dict[str, Any],
        career_state: Any  # EmployeeCareerState or dict
    ) -> Dict[str, str]:
        """
        Generate personalized pay explanation text.
        
        This method generates TEXT ONLY. All numeric values must be provided in pay_breakdown
        and compliance_data from deterministic services (PayCalculationService, ReleaseEngine).
        
        Args:
            employee: User instance (employee)
            pay_breakdown: Dict from PayCalculationService/ReleaseEngine with:
                - earned_amount: Decimal
                - released_amount: Decimal
                - locked_amount: Decimal
                - stipend: Decimal
                - safety_net_topup: Decimal
                - base_release: Decimal
            compliance_data: Dict from ComplianceCalculator with:
                - completion_rate: float
                - is_compliant: bool
                - threshold: float
            career_state: EmployeeCareerState instance or dict with group, is_tenured, etc.
        
        Returns:
            Dict with keys:
                - kes: str (English explanation in KES context)
                - swahili: Optional[str] (Swahili translation for Group B employees)
                - reasoning_steps: List[str] (chain-of-thought steps, if available)
        """
        if not self.ai_facade or not self.prompt_registry:
            # Fallback to template-based explanation
            return self._fallback_pay_explanation(pay_breakdown, compliance_data)
        
        try:
            # Determine which prompt template to use based on compliance and pay status
            if compliance_data.get('is_compliant', False):
                template_name = "pay_explanation.compliance_unlocked"
            else:
                template_name = "pay_explanation.compliance_locked"
            
            # Build context for prompt
            context = {
                'employee_name': f"{employee.first_name} {employee.last_name}".strip() or employee.username,
                'compliance_rate': compliance_data.get('completion_rate', 0.0),
                'is_compliant': compliance_data.get('is_compliant', False),
                'threshold': compliance_data.get('threshold', 33.0),
                'earned_amount': float(pay_breakdown.get('earned_amount', 0)),
                'released_amount': float(pay_breakdown.get('released_amount', 0)),
                'locked_amount': float(pay_breakdown.get('locked_amount', 0)),
                'stipend': float(pay_breakdown.get('stipend', 0)),
                'safety_net_topup': float(pay_breakdown.get('safety_net_topup', 0)),
                'base_release': float(pay_breakdown.get('base_release', 0)),
            }
            
            # Get career state info
            if hasattr(career_state, 'group'):
                context['employee_group'] = career_state.group
            elif isinstance(career_state, dict):
                context['employee_group'] = career_state.get('group', 'B')
            else:
                context['employee_group'] = 'B'
            
            # Render prompt
            prompt = self.prompt_registry.render(template_name, context)
            
            # TODO: Call AI service with prompt and validate output against PayExplanationOutput schema
            # For now, return placeholder
            return {
                'kes': f"AI-generated explanation placeholder for {employee.username}",
                'swahili': None,
                'reasoning_steps': []
            }
            
        except Exception as e:
            self.logger.error(f"Error generating pay explanation for {employee.username}: {e}", exc_info=True)
            return self._fallback_pay_explanation(pay_breakdown, compliance_data)
    
    def generate_daf_focus(
        self,
        employee: User,
        career_data: Dict[str, Any],
        money_data: Dict[str, Any],
        metrics_data: Dict[str, Any],
        activities_data: List[Dict[str, Any]]
    ) -> Dict[str, Any]:
        """
        Generate personalized DAF focus recommendations.
        
        This method generates TEXT RECOMMENDATIONS ONLY. All data must be provided from
        deterministic services (CareerLevelService, PayCalculationService, PerformanceMetricsService).
        
        Args:
            employee: User instance (employee)
            career_data: Dict from CareerLevelService with current_level, next_level, progress, blockers
            money_data: Dict from PayCalculationService with earned, released, locked amounts
            metrics_data: Dict from PerformanceMetricsService with compliance, quality, attendance
            activities_data: List of activity dicts from DAFSummaryService
        
        Returns:
            Dict with keys:
                - focus_areas: List[str]
                - recommendations: List[str]
                - action_items: List[Dict[str, str]] (e.g., [{"task": "...", "deadline": "..."}])
                - estimated_impact: Optional[str]
                - reasoning: List[str] (chain-of-thought steps)
        """
        if not self.ai_facade or not self.prompt_registry:
            return self._fallback_daf_focus(metrics_data)
        
        try:
            # Determine focus template based on metrics
            compliance = metrics_data.get('compliance', 0.0)
            quality_score = metrics_data.get('checklist_quality_score', 0.0)
            
            if compliance < 33.0:
                template_name = "daf_focus.low_compliance"
            elif quality_score < 0.5:
                template_name = "daf_focus.low_quality"
            else:
                template_name = "daf_focus.missing_evidence"  # Default
            
            # Build context
            context = {
                'employee_name': f"{employee.first_name} {employee.last_name}".strip() or employee.username,
                'compliance': compliance,
                'quality_score': quality_score,
                'attendance_days': metrics_data.get('attendance_days', 0),
                'current_level': career_data.get('current_level_code', 'B1'),
                'next_level': career_data.get('next_level', {}).get('code', 'B2'),
                'progress': career_data.get('progress_to_next', 0.0),
                'blockers': career_data.get('blockers', []),
                'locked_amount': money_data.get('locked_amount', 0),
            }
            
            # Render prompt
            prompt = self.prompt_registry.render(template_name, context)
            
            # TODO: Call AI service, validate against DAFFocusOutput schema
            return {
                'focus_areas': ["AI-generated focus placeholder"],
                'recommendations': ["AI-generated recommendation placeholder"],
                'action_items': [],
                'estimated_impact': None,
                'reasoning': []
            }
            
        except Exception as e:
            self.logger.error(f"Error generating DAF focus for {employee.username}: {e}", exc_info=True)
            return self._fallback_daf_focus(metrics_data)
    
    def generate_career_coaching(
        self,
        employee: User,
        career_state: Dict[str, Any],
        progress: Dict[str, Any],
        blockers: List[str]
    ) -> Dict[str, str]:
        """
        Generate personalized career coaching text.
        
        This method generates TEXT ONLY. All career level data must be provided from
        deterministic services (CareerLevelService).
        
        Args:
            employee: User instance (employee)
            career_state: Dict from CareerLevelService.get_state() with current_level, group, etc.
            progress: Dict from CareerLevelService.get_progress_to_next_level() with progress, next_level, etc.
            blockers: List of blocker strings from CareerLevelService
        
        Returns:
            Dict with keys:
                - summary: str (overall coaching summary)
                - blockers: List[str] (explanation of blockers)
                - next_steps: List[str] (actionable next steps)
                - estimated_months_to_next_level: Optional[float] (from deterministic service)
        """
        if not self.ai_facade or not self.prompt_registry:
            return self._fallback_career_coaching(progress, blockers)
        
        try:
            # Determine template
            if blockers:
                template_name = "career_coaching.blockers"
            elif progress.get('estimated_months'):
                template_name = "career_coaching.estimated_time"
            else:
                template_name = "career_coaching.progress_update"
            
            # Build context
            context = {
                'employee_name': f"{employee.first_name} {employee.last_name}".strip() or employee.username,
                'current_level': career_state.get('current_level_code', 'B1'),
                'next_level': progress.get('next_level', {}).get('code', 'B2'),
                'progress': progress.get('progress', 0.0),
                'blockers': blockers,
                'estimated_months': progress.get('estimated_months'),
            }
            
            # Render prompt
            prompt = self.prompt_registry.render(template_name, context)
            
            # TODO: Call AI service, validate against CareerCoachingOutput schema
            return {
                'summary': f"AI-generated career coaching placeholder for {employee.username}",
                'blockers': blockers,
                'next_steps': ["AI-generated next step placeholder"],
                'estimated_months_to_next_level': progress.get('estimated_months')
            }
            
        except Exception as e:
            self.logger.error(f"Error generating career coaching for {employee.username}: {e}", exc_info=True)
            return self._fallback_career_coaching(progress, blockers)
    
    def generate_quality_feedback(
        self,
        employee: User,
        task: Any,  # Task model instance
        quality_score: float,
        missing_items: List[str]
    ) -> str:
        """
        Generate personalized quality feedback text for a task.
        
        This method generates TEXT ONLY. Quality score must be provided from
        deterministic services (ChecklistEvaluationService).
        
        Args:
            employee: User instance (employee)
            task: Task model instance
            quality_score: float (0-1) from ChecklistEvaluationService
            missing_items: List of missing checklist items
        
        Returns:
            str: Personalized quality feedback message
        """
        if not self.ai_facade or not self.prompt_registry:
            return self._fallback_quality_feedback(quality_score, missing_items)
        
        try:
            # Determine template based on quality score
            if quality_score >= 0.8:
                template_name = "quality_feedback.high_quality"
            elif quality_score >= 0.5:
                template_name = "quality_feedback.medium_quality"
            else:
                template_name = "quality_feedback.low_quality"
            
            # Build context
            context = {
                'employee_name': f"{employee.first_name} {employee.last_name}".strip() or employee.username,
                'task_name': getattr(task, 'activity_name', 'Task'),
                'quality_score': quality_score,
                'quality_percentage': quality_score * 100,
                'missing_items': missing_items,
            }
            
            # Render prompt
            prompt = self.prompt_registry.render(template_name, context)
            
            # TODO: Call AI service, validate against QualityFeedbackOutput schema
            return f"AI-generated quality feedback placeholder for {task.activity_name if hasattr(task, 'activity_name') else 'task'}"
            
        except Exception as e:
            self.logger.error(f"Error generating quality feedback for {employee.username}: {e}", exc_info=True)
            return self._fallback_quality_feedback(quality_score, missing_items)
    
    def generate_compliance_coaching(
        self,
        employee: User,
        compliance_data: Dict[str, Any],
        target_month: int,
        target_year: int
    ) -> Dict[str, str]:
        """
        Generate personalized compliance coaching text.
        
        This method generates TEXT ONLY. Compliance data must be provided from
        deterministic services (ComplianceCalculator).
        
        Args:
            employee: User instance (employee)
            compliance_data: Dict from ComplianceCalculator with:
                - completion_rate: float
                - is_compliant: bool
                - total_points: Decimal
                - total_max_points: Decimal
                - threshold: float
            target_month: Month being coached (1-12)
            target_year: Year being coached
        
        Returns:
            Dict with keys:
                - kes: str (English coaching message)
                - swahili: Optional[str] (Swahili translation for Group B)
                - action_items: List[str] (specific actions to take)
        """
        if not self.ai_facade or not self.prompt_registry:
            return self._fallback_compliance_coaching(compliance_data)
        
        try:
            # Build context
            context = {
                'employee_name': f"{employee.first_name} {employee.last_name}".strip() or employee.username,
                'completion_rate': compliance_data.get('completion_rate', 0.0),
                'is_compliant': compliance_data.get('is_compliant', False),
                'threshold': compliance_data.get('threshold', 33.0),
                'total_points': float(compliance_data.get('total_points', 0)),
                'total_max_points': float(compliance_data.get('total_max_points', 0)),
                'target_month': target_month,
                'target_year': target_year,
            }
            
            # Calculate gap if not compliant
            if not compliance_data.get('is_compliant', False):
                gap = compliance_data.get('threshold', 33.0) - compliance_data.get('completion_rate', 0.0)
                context['gap_percentage'] = gap
                # Estimate tasks needed (rough calculation)
                if compliance_data.get('total_max_points', 0) > 0:
                    points_needed = (gap / 100.0) * float(compliance_data.get('total_max_points', 0))
                    context['points_needed'] = points_needed
            
            # Render prompt (use daf_focus.low_compliance as base)
            prompt = self.prompt_registry.render("daf_focus.low_compliance", context)
            
            # TODO: Call AI service, generate coaching message
            return {
                'kes': f"AI-generated compliance coaching placeholder for {employee.username}",
                'swahili': None,
                'action_items': ["AI-generated action item placeholder"]
            }
            
        except Exception as e:
            self.logger.error(f"Error generating compliance coaching for {employee.username}: {e}", exc_info=True)
            return self._fallback_compliance_coaching(compliance_data)
    
    # Fallback methods (template-based, no AI)
    
    def _fallback_pay_explanation(
        self,
        pay_breakdown: Dict[str, Any],
        compliance_data: Dict[str, Any]
    ) -> Dict[str, str]:
        """Fallback template-based pay explanation when AI is unavailable."""
        compliance_rate = compliance_data.get('completion_rate', 0.0)
        if compliance_data.get('is_compliant', False):
            kes = f"Your current month compliance is {compliance_rate:.1f}%, so you unlocked your earnings."
        else:
            kes = f"You are at {compliance_rate:.1f}% completion. You need at least 33% to unlock your earnings."
        
        return {
            'kes': kes,
            'swahili': None,
            'reasoning_steps': []
        }
    
    def _fallback_daf_focus(self, metrics_data: Dict[str, Any]) -> Dict[str, Any]:
        """Fallback template-based DAF focus when AI is unavailable."""
        compliance = metrics_data.get('compliance', 0.0)
        if compliance < 33.0:
            focus_areas = ["Focus on completing more tasks this month"]
        else:
            focus_areas = ["Maintain your current performance"]
        
        return {
            'focus_areas': focus_areas,
            'recommendations': ["Continue working on your tasks"],
            'action_items': [],
            'estimated_impact': None,
            'reasoning': []
        }
    
    def _fallback_career_coaching(
        self,
        progress: Dict[str, Any],
        blockers: List[str]
    ) -> Dict[str, str]:
        """Fallback template-based career coaching when AI is unavailable."""
        return {
            'summary': f"You're making progress toward {progress.get('next_level', {}).get('code', 'next level')}",
            'blockers': blockers,
            'next_steps': ["Continue completing tasks with high quality"],
            'estimated_months_to_next_level': progress.get('estimated_months')
        }
    
    def _fallback_quality_feedback(
        self,
        quality_score: float,
        missing_items: List[str]
    ) -> str:
        """Fallback template-based quality feedback when AI is unavailable."""
        if quality_score >= 0.8:
            return "Great work! Your quality score is excellent."
        elif quality_score >= 0.5:
            return f"Your quality score is {quality_score * 100:.0f}%. To improve, add evidence for: {', '.join(missing_items)}"
        else:
            return f"Your quality score is {quality_score * 100:.0f}%. Please add evidence for: {', '.join(missing_items)}"
    
    def _fallback_compliance_coaching(self, compliance_data: Dict[str, Any]) -> Dict[str, str]:
        """Fallback template-based compliance coaching when AI is unavailable."""
        compliance_rate = compliance_data.get('completion_rate', 0.0)
        threshold = compliance_data.get('threshold', 33.0)
        gap = threshold - compliance_rate
        
        if gap > 0:
            kes = f"You're at {compliance_rate:.1f}% completion. You need {gap:.1f}% more to reach the {threshold}% threshold."
        else:
            kes = f"Great! You've reached {compliance_rate:.1f}% completion, exceeding the {threshold}% threshold."
        
        return {
            'kes': kes,
            'swahili': None,
            'action_items': ["Complete more tasks this month"]
        }


# Global instance helper
def get_ai_insight_service() -> Optional[AIInsightService]:
    """
    Get or create global AIInsightService instance.

    """
    try:
        return AIInsightService()
    except Exception as e:

"""
Prompt Template Registry

Manages all AI prompt templates with versioning and variable validation.

Templates are stored in-code for now, but can be migrated to database or config files later.
"""

from dataclasses import dataclass
from typing import Dict, List, Optional, Any
import logging
import re

logger = logging.getLogger(__name__)


@dataclass
class PromptTemplate:
    """
    Represents a prompt template with metadata.
    
    Attributes:
        name: Unique template name (e.g., "pay_explanation.compliance_unlocked")
        version: Template version string (default: "1.0")
        template: Jinja2-style template string with {{ variable }} placeholders
        variables: List of required variable names
        system_prompt: Optional system prompt for AI model
        temperature: Temperature setting for AI model (default: 0.3)
        output_schema_name: Optional output schema name for validation
    """
    name: str
    version: str
    template: str
    variables: List[str]
    system_prompt: Optional[str] = None
    temperature: float = 0.3
    output_schema_name: Optional[str] = None
    
    def __post_init__(self):
        """Validate template after initialization."""
        # Extract variables from template
        found_vars = re.findall(r'\{\{\s*(\w+)\s*\}\}', self.template)
        missing_vars = set(found_vars) - set(self.variables)
        if missing_vars:
            logger.warning(f"Template {self.name} uses variables not in variables list: {missing_vars}")

        Args:
            name: Template name (e.g., "pay_explanation.compliance_unlocked")
            version: Template version (default: "latest" - returns most recent)
        
        Returns:
            PromptTemplate instance or None if not found
        """
        # For now, version is ignored (all templates are version "1.0")
        # TODO: Implement version tracking when migrating to database
        return self._templates.get(name)
    
    def render(self, name: str, context: Dict[str, Any], version: str = "latest") -> str:
        """
        Render a template with the given context.
        
        Args:
            name: Template name
            context: Dictionary of variables to substitute
            version: Template version (default: "latest")
        
        Returns:
            Rendered template string
        
        Raises:
            ValueError: If template not found or required variables missing
        """
        template = self.get_template(name, version)
        if not template:
            raise ValueError(f"Template '{name}' not found")
        
        # Simple template rendering (Jinja2-style but basic implementation)
        # TODO: Replace with actual Jinja2 when adding dependency
        rendered = template.template
        
        # Replace {{ variable }} placeholders
        for var in template.variables:
            if var not in context:
                logger.warning(f"Variable '{var}' not provided in context for template '{name}'")
                continue
            value = context[var]
            # Format based on type
            if isinstance(value, float):
                formatted_value = f"{value:.1f}" if value % 1 != 0 else f"{int(value)}"
            elif isinstance(value, (list, tuple)):
                formatted_value = ", ".join(str(item) for item in value)
            else:
                formatted_value = str(value)
            rendered = rendered.replace(f"{{{{ {var} }}}}", formatted_value)
        
        # Handle simple conditionals {% if ... %}...{% endif %}
        # TODO: Replace with actual Jinja2 when adding dependency
        import re
        conditional_pattern = r'\{%\s*if\s+(\w+)\s*%\}\s*(.*?)\s*\{%\s*endif\s*%\}'
        for match in re.finditer(conditional_pattern, rendered, re.DOTALL):
            var_name = match.group(1)
            content = match.group(2)
            if var_name in context and context[var_name]:
                rendered = rendered.replace(match.group(0), content)
            else:
                rendered = rendered.replace(match.group(0), "")
        
        # Handle filters like |join(', ')
        # TODO: Replace with actual Jinja2 when adding dependency
        filter_pattern = r'(\w+)\s*\|\s*join\s*\(\s*[\'"]?([^\'"]*)[\'"]?\s*\)'
        for match in re.finditer(filter_pattern, rendered):
            var_name = match.group(1)
            separator = match.group(2) or ", "
            if var_name in context and isinstance(context[var_name], (list, tuple)):
                formatted_list = separator.join(str(item) for item in context[var_name])
                rendered = rendered.replace(match.group(0), formatted_list)
        
        return rendered
    
    def list_templates(self) -> List[PromptTemplate]:
        """
        List all registered templates.
        
        Returns:

        return list(self._templates.values())

    """
    Registry for managing output schemas.
    
    Maps output type names to Pydantic model classes for validation.
    """
    
    def __init__(self):
        """Initialize registry with default schemas."""
        self._schemas: Dict[str, Type[BaseModel]] = {
            "pay_explanation": PayExplanationOutput,
            "daf_focus": DAFFocusOutput,
            "career_coaching": CareerCoachingOutput,
            "quality_feedback": QualityFeedbackOutput,
        }
    
    def get_schema(self, output_type: str) -> Optional[Type[BaseModel]]:
        """
        Get schema class for an output type.
        
        Args:
            output_type: Output type name (e.g., "pay_explanation")
        
        Returns:
            Pydantic BaseModel class or None if not found
        """
        return self._schemas.get(output_type)
    
    def validate(self, output_type: str, data: Dict[str, Any]) -> BaseModel:
        """
        Validate data against a schema.
        
        Args:
            output_type: Output type name
            data: Data dictionary to validate
        
        Returns:
            Validated Pydantic model instance
        
        Raises:
            ValueError: If output_type not found
            ValidationError: If data doesn't conform to schema
        """
        schema_class = self.get_schema(output_type)
        if not schema_class:
            raise ValueError(f"Output schema '{output_type}' not found")
        
        try:
            return schema_class(**data)
        except Exception as e:
            logger.error(f"Validation failed for {output_type}: {e}", exc_info=True)
            raise
    
    def register_schema(self, output_type: str, schema_class: Type[BaseModel]):
        """
        Register a new output schema.
        
        Args:
            output_type: Output type name
            schema_class: Pydantic BaseModel class
        """
        self._schemas[output_type] = schema_class
        logger.info(f"Registered output schema: {output_type}")

"""
Activity Definition Model

AI metadata for ActivityType definitions.

This model extends ActivityType with AI-specific context and templates
without modifying the core ActivityType model or breaking existing migrations.
"""

from django.db import models
from shared_core.models import TimeStampedModel
from django.utils.translation import gettext_lazy as _


class ActivityDefinition(TimeStampedModel):
    """
    AI metadata and context for ActivityType definitions.
    
    This model provides AI-specific information for each activity type:
    - AI context descriptions for prompts
    - Prompt template names
    - Quality criteria and thresholds
    - Coaching templates
    - Skill requirements for intelligent assignment
    - Evidence requirements
    
    One-to-one relationship with ActivityType - each ActivityType can have
    one ActivityDefinition (optional).
    """
    
    activity_type = models.OneToOneField(
        'management.ActivityType',
        on_delete=models.CASCADE,
        related_name='ai_definition',
        help_text=_("The ActivityType this definition extends")
    )
    
    ai_context = models.TextField(
        blank=True,
        help_text=_("High-level description of what this activity means in CODA, for AI prompts. "
                   "This helps AI understand the activity's purpose and context.")
    )
    
    prompt_template_name = models.CharField(
        max_length=255,
        blank=True,
        help_text=_("Default prompt template name for this activity (e.g., 'activity.bi_session.assignment_prompt')")
    )
    
    quality_criteria = models.JSONField(
        default=dict,
        blank=True,
        help_text=_("JSON dict defining what constitutes 'good' evidence for this activity. "
                   "Example: {'requires_photos': True, 'min_evidence_items': 3}")
    )
    
    coaching_template_name = models.CharField(
        max_length=255,
        blank=True,
        help_text=_("Default coaching template name for this activity (e.g., 'activity.bi_session.coaching_prompt')")
    )
    
    skill_requirements = models.JSONField(
        default=list,
        blank=True,
        help_text=_("List of skills required for this activity (for intelligent assignment). "
                   "Example: ['data_analysis', 'sql', 'reporting']")
    )
    
    evidence_requirements = models.JSONField(
        default=list,
        blank=True,
        help_text=_("List of required evidence types for this activity. "
                   "Example: ['before_photos', 'after_photos', 'summary_notes']")
    )
    
    quality_thresholds = models.JSONField(
        default=dict,
        blank=True,
        help_text=_("JSON dict defining quality score thresholds for this activity. "
                   "Example: {'promotion_weight': 'high', 'min_quality_for_promotion': 0.8}")
    )
    
    class Meta:
        verbose_name = "Activity Definition"
        verbose_name_plural = "Activity Definitions"
        ordering = ['activity_type__name']
    
    def __str__(self):
        return f"AI Definition for {self.activity_type.name if self.activity_type else 'Unknown Activity'}"
